# kunPet Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 交付一个 Cursor/VS Code 扩展：激活后常驻鲲桌宠悬浮窗；Agent `stop` 时播放庆祝动画并弹出系统通知。

**Architecture:** 扩展宿主负责 Hook 注册、本机 HTTP 事件服务、系统通知与子进程管理；随扩展分发的 Electron 小窗负责透明置顶渲染；Cursor `stop` Hook 向扩展 POST `agent_stop`，扩展再经 IPC 通知桌宠并 Toast。

**Tech Stack:** TypeScript、VS Code Extension API、Node `http`、Electron（桌宠子进程）、Cursor `hooks.json`、Windows 优先。

## Global Constraints

- 仅一个扩展安装入口；桌宠二进制随扩展分发，不另发独立安装包
- 扩展激活后桌宠默认一直显示（可手动隐藏）
- 完成信号：Cursor Agent `stop` only
- Hook/桌宠失败必须静默，不得影响 Agent（退出码 0）
- HTTP 仅监听 `127.0.0.1`；扩展为唯一端口监听方
- 通知文案：标题「鲲来报喜」，正文「这轮 AI 对话完成啦」
- 首版不做养成/闲逛；Windows 优先
- 规格来源：`docs/superpowers/specs/2026-08-19-kunpet-design.md`

```typescript
import http from "node:http";
import type { AgentStopEvent } from "./types";
import { DEDUPE_WINDOW_MS } from "./types";

export function shouldDedupe(now: number, lastTs: number, windowMs: number): boolean {
  if (!lastTs) return false;
  return now - lastTs < windowMs;
}

export async function startEventServer(opts: {
  onAgentStop: (e: AgentStopEvent) => void;
  preferredPort?: number;
}): Promise<{ port: number; close: () => Promise<void> }> {
  let lastAcceptedTs = 0;

  const server = http.createServer((req, res) => {
    if (req.method === "POST" && req.url === "/event") {
      const chunks: Buffer[] = [];
      req.on("data", (c) => chunks.push(c));
      req.on("end", () => {
        try {
          const body = JSON.parse(Buffer.concat(chunks).toString("utf8")) as AgentStopEvent;
          if (body?.type !== "agent_stop" || typeof body.ts !== "number") {
            res.writeHead(400).end("bad request");
            return;
          }
          const now = Date.now();
          if (shouldDedupe(now, lastAcceptedTs, DEDUPE_WINDOW_MS)) {
            res.writeHead(200).end("deduped");
            return;
          }
          lastAcceptedTs = now;
          opts.onAgentStop(body);
          res.writeHead(200).end("ok");
        } catch {
          res.writeHead(400).end("bad json");
        }
      });
      return;
    }
    res.writeHead(404).end("not found");
  });

  const port = await new Promise<number>((resolve, reject) => {
    const tryListen = (p: number) => {
      const onError = (err: NodeJS.ErrnoException) => {
        server.off("error", onError);
        if (err.code === "EADDRINUSE" && p < 19300) {
          tryListen(p + 1);
        } else {
          reject(err);
        }
      };
      server.once("error", onError);
      server.listen(p, "127.0.0.1", () => {
        server.off("error", onError);
        const addr = server.address();
        if (addr && typeof addr === "object") resolve(addr.port);
        else reject(new Error("no address"));
      });
    };
    tryListen(opts.preferredPort ?? 19246);
  });

  return {
    port,
    close: () =>
      new Promise((resolve, reject) => {
        server.close((err) => (err ? reject(err) : resolve()));
      }),
  };
}
```

```typescript
import fs from "node:fs";
import fsp from "node:fs/promises";
import os from "node:os";
import path from "node:path";
import {
  HOOK_SCRIPT_NAME,
  PORT_FILE_NAME,
  type PortFileContents,
} from "./types";

type HooksFile = {
  version?: number;
  hooks?: Record<string, Array<{ command?: string; [k: string]: unknown }>>;
};

export function getCursorHome(): string {
  return path.join(os.homedir(), ".cursor");
}

export function toHookPath(absPath: string): string {
  return absPath.replace(/\\/g, "/");
}

export function writePortFile(cursorHome: string, port: number): void {
  const file = path.join(cursorHome, PORT_FILE_NAME);
  const body: PortFileContents = { port, updatedAt: Date.now() };
  fs.mkdirSync(cursorHome, { recursive: true });
  fs.writeFileSync(file, JSON.stringify(body), "utf8");
}

export function readPortFile(cursorHome: string): PortFileContents | null {
  const file = path.join(cursorHome, PORT_FILE_NAME);
  if (!fs.existsSync(file)) return null;
  try {
    return JSON.parse(fs.readFileSync(file, "utf8")) as PortFileContents;
  } catch {
    return null;
  }
}

export function mergeStopHook(hooksJson: unknown, command: string): HooksFile {
  const base: HooksFile =
    hooksJson && typeof hooksJson === "object"
      ? (JSON.parse(JSON.stringify(hooksJson)) as HooksFile)
      : {};
  base.version = base.version ?? 1;
  base.hooks = base.hooks ?? {};
  const stop = Array.isArray(base.hooks.stop) ? base.hooks.stop : [];
  const exists = stop.some(
    (h) => typeof h?.command === "string" && h.command.includes(HOOK_SCRIPT_NAME)
  );
  if (!exists) stop.push({ command });
  base.hooks.stop = stop;
  return base;
}

export function removeStopHook(hooksJson: unknown, marker: string): HooksFile {
  const base: HooksFile =
    hooksJson && typeof hooksJson === "object"
      ? (JSON.parse(JSON.stringify(hooksJson)) as HooksFile)
      : { version: 1, hooks: {} };
  base.hooks = base.hooks ?? {};
  const stop = Array.isArray(base.hooks.stop) ? base.hooks.stop : [];
  base.hooks.stop = stop.filter(
    (h) => !(typeof h?.command === "string" && h.command.includes(marker))
  );
  return base;
}

export async function installHookScript(
  cursorHome: string,
  scriptSourcePath: string
): Promise<string> {
  const hooksDir = path.join(cursorHome, "hooks");
  await fsp.mkdir(hooksDir, { recursive: true });
  const dest = path.join(hooksDir, HOOK_SCRIPT_NAME);
  await fsp.copyFile(scriptSourcePath, dest);
  return toHookPath(dest);
}

export async function ensureKunPetHook(opts: {
  extensionHookSource: string;
  port: number;
}): Promise<void> {
  const cursorHome = getCursorHome();
  writePortFile(cursorHome, opts.port);
  const dest = await installHookScript(cursorHome, opts.extensionHookSource);
  const command = `node "${dest}"`;
  const hooksPath = path.join(cursorHome, "hooks.json");
  let current: unknown = { version: 1, hooks: {} };
  if (fs.existsSync(hooksPath)) {
    current = JSON.parse(await fsp.readFile(hooksPath, "utf8"));
  }
  const merged = mergeStopHook(current, command);
  await fsp.writeFile(hooksPath, JSON.stringify(merged, null, 2), "utf8");
}

export async function cleanupKunPetHook(): Promise<void> {
  const cursorHome = getCursorHome();
  const hooksPath = path.join(cursorHome, "hooks.json");
  if (!fs.existsSync(hooksPath)) return;
  const current = JSON.parse(await fsp.readFile(hooksPath, "utf8"));
  const next = removeStopHook(current, HOOK_SCRIPT_NAME);
  await fsp.writeFile(hooksPath, JSON.stringify(next, null, 2), "utf8");
}
```

- [ ] **Step 4: 实现 `extension/hooks/kunpet-notify.js`（纯 JS，无构建）**

```javascript
#!/usr/bin/env node
/**
 * Cursor stop hook: notify kunPet extension. Always exit 0.
 */
const fs = require("fs");
const http = require("http");
const os = require("os");
const path = require("path");

function failOpen() {
  process.exit(0);
}

try {
  const portFile = path.join(os.homedir(), ".cursor", "kunpet-port.json");
  if (!fs.existsSync(portFile)) failOpen();
  const { port } = JSON.parse(fs.readFileSync(portFile, "utf8"));
  if (!port) failOpen();

  const body = JSON.stringify({ type: "agent_stop", ts: Date.now() });
  const req = http.request(
    {
      host: "127.0.0.1",
      port,
      path: "/event",
      method: "POST",
      headers: {
        "Content-Type": "application/json",
        "Content-Length": Buffer.byteLength(body),
      },
      timeout: 800,
    },
    (res) => {
      res.resume();
      res.on("end", failOpen);
    }
  );
  req.on("error", failOpen);
  req.on("timeout", () => {
    req.destroy();
    failOpen();
  });
  req.write(body);
  req.end();
} catch {
  failOpen();
}
```

```javascript
const { app, BrowserWindow, Tray, Menu, nativeImage } = require("electron");
const path = require("path");
const readline = require("readline");

let win;
let tray;

function argNum(name, fallback) {
  const hit = process.argv.find((a) => a.startsWith(`--${name}=`));
  if (!hit) return fallback;
  const n = Number(hit.split("=")[1]);
  return Number.isFinite(n) ? n : fallback;
}

function createWindow() {
  const x = argNum("x", undefined);
  const y = argNum("y", undefined);
  win = new BrowserWindow({
    width: 128,
    height: 128,
    x,
    y,
    frame: false,
    transparent: true,
    alwaysOnTop: true,
    resizable: false,
    skipTaskbar: true,
    hasShadow: false,
    webPreferences: {
      preload: path.join(__dirname, "preload.js"),
      contextIsolation: true,
      nodeIntegration: false,
    },
  });
  if (process.platform === "win32") {
    try {
      win.setAlwaysOnTop(true, "screen-saver");
    } catch (_) {
      win.setAlwaysOnTop(true);
    }
  }
  win.loadFile(path.join(__dirname, "renderer", "index.html"));
  win.setIgnoreMouseEvents(false);
  win.on("moved", () => {
    const [wx, wy] = win.getPosition();
    process.stdout.write(JSON.stringify({ type: "moved", x: wx, y: wy }) + "\n");
  });
}

function handleMessage(msg) {
  if (!win) return;
  switch (msg.type) {
    case "celebrate":
      win.showInactive();
      win.webContents.send("pet:celebrate");
      break;
    case "show":
      win.showInactive();
      break;
    case "hide":
      win.hide();
      break;
    case "set-position":
      win.setPosition(msg.x, msg.y);
      break;
  }
}

function setupStdin() {
  const rl = readline.createInterface({ input: process.stdin });
  rl.on("line", (line) => {
    try {
      handleMessage(JSON.parse(line));
    } catch (_) {}
  });
}

function setupTray() {
  const img = nativeImage.createEmpty();
  tray = new Tray(img);
  tray.setToolTip("kunPet");
  tray.setContextMenu(
    Menu.buildFromTemplate([
      { label: "显示", click: () => win?.showInactive() },
      { label: "隐藏", click: () => win?.hide() },
    ])
  );
}

app.whenReady().then(() => {
  createWindow();
  setupTray();
  setupStdin();
  process.stdout.write(JSON.stringify({ type: "ready" }) + "\n");
});
```

```typescript
function resolvePetRoot(extensionPath: string): string {
  const bundled = path.join(extensionPath, "pet");
  const sibling = path.join(extensionPath, "..", "pet");
  return fs.existsSync(path.join(bundled, "main.js")) ? bundled : sibling;
}
```
